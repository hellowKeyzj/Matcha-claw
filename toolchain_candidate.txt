165	/// Foundation-backed native command execution for the Toolchain owner.
166	///
167	/// The child receives only the process environment needed by uv and is always launched through
168	/// Foundation's contained one-shot runner. No command output crosses this seam.
169	pub struct FoundationToolchainCommandPort {
170	    working_directory: PathBuf,
171	    public_environment: Vec<(OsString, OsString)>,
172	    #[cfg(unix)]
173	    guardian_executable: PathBuf,
174	}

176	impl FoundationToolchainCommandPort {
177	    #[cfg(windows)]
178	    pub fn new(working_directory: PathBuf) -> Self {
179	        Self {
180	            working_directory,
181	            public_environment: toolchain_environment(),
182	        }
183	    }
184	
185	    #[cfg(unix)]
186	    pub fn new(working_directory: PathBuf, guardian_executable: PathBuf) -> Self {
187	        Self {
188	            working_directory,
189	            public_environment: toolchain_environment(),
190	            guardian_executable,
191	        }
192	    }
193	
194	    fn environment_value(&self, key: &OsStr) -> Option<&OsStr> {
195	        self.public_environment
196	            .iter()
197	            .find(|(name, _)| name == key)
198	            .map(|(_, value)| value.as_os_str())
199	    }
200	
201	    fn resolve_path(&self, program: &Path) -> Option<PathBuf> {
202	        if program.is_absolute() {
203	            return is_file(program).then(|| program.to_owned());
204	        }
205	
206	        let name = program.file_name()?;
207	        #[cfg(windows)]
208	        if name.eq_ignore_ascii_case(OsStr::new("where.exe")) {
209	            if let Some(root) = self.environment_value(OsStr::new("SystemRoot")) {
210	                let where_path = PathBuf::from(root).join("System32").join("where.exe");
211	                if let Some(path) = is_file(&where_path).then_some(where_path) {
212	                    return Some(path);
213	                }
214	            }
215	        }
216	
217	        let path = self.environment_value(OsStr::new("PATH"))?;
218	        for entry in std::env::split_paths(path) {
219	            let directory = if entry.as_os_str().is_empty() {
220	                self.working_directory.clone()
221	            } else if entry.is_absolute() {
222	                entry
223	            } else {
224	                self.working_directory.join(entry)
225	            };
226	            let candidate = directory.join(name);
227	            if let Some(path) = is_file(&candidate).then_some(candidate) {
228	                return Some(path);
229	            }
230	            #[cfg(windows)]
231	            if Path::new(name).extension().is_none() {
232	                let candidate = directory.join(format!("{}.exe", name.to_string_lossy()));
233	                if let Some(path) = is_file(&candidate).then_some(candidate) {
234	                    return Some(path);
235	                }
236	            }
237	        }
238	        None
239	    }
240	}

242	impl ToolchainCommandPort for FoundationToolchainCommandPort {
243	    fn execute(&self, request: ToolchainCommandRequest) -> ToolchainCommandFuture {
244	        let Some(program) = self.resolve_path(request.program()) else {
245	            return Box::pin(async { ToolchainCommandOutcome::Unavailable });
246	        };
247	        let launch = LaunchSpec::try_new(
248	            program,
249	            self.working_directory.clone(),
250	            request.arguments().to_owned(),
251	            self.public_environment.clone(),
252	            StdioSpec::new(StdioMode::Null, StdioMode::Null, StdioMode::Null),
253	        );
254	        let launch = match launch {
255	            Ok(launch) => launch,
256	            Err(_) => return Box::pin(async { ToolchainCommandOutcome::Unavailable }),
257	        };
258	        #[cfg(windows)]
259	        let containment = OneShotContainment::job();
260	        #[cfg(unix)]
261	        let containment = match OneShotContainment::guardian(self.guardian_executable.clone()) {
262	            Ok(containment) => containment,
263	            Err(_) => return Box::pin(async { ToolchainCommandOutcome::Unavailable }),
264	        };
265	        let deadline = request.timeout().unwrap_or(INSTALL_TIMEOUT);
266	        let run = match OneShotRun::try_new(launch, containment, deadline) {
267	            Ok(run) => run,
268	            Err(_) => return Box::pin(async { ToolchainCommandOutcome::Unavailable }),
269	        };
270	        let mut operation = match OneShotRunner::new().start(run) {
271	            Ok(operation) => operation,
272	            Err(_) => return Box::pin(async { ToolchainCommandOutcome::Unavailable }),
273	        };
274	        Box::pin(async move { map_one_shot_completion(operation.wait().await) })
275	    }
276	
277	    fn resolve_program(&self, program: &Path) -> Option<PathBuf> {
278	        self.resolve_path(program)
279	    }
280	}

282	fn is_file(path: &Path) -> bool {
283	    fs::metadata(path).is_ok_and(|metadata| metadata.is_file())
284	}

286	fn toolchain_environment() -> Vec<(OsString, OsString)> {
287	    const KEYS: [&str; 10] = [
288	        "PATH",
289	        "HOME",
290	        "USERPROFILE",
291	        "HOMEDRIVE",
292	        "HOMEPATH",
293	        "LOCALAPPDATA",
294	        "APPDATA",
295	        "TEMP",
296	        "TMP",
297	        "XDG_CACHE_HOME",
298	    ];
299	    let mut environment = KEYS
300	        .into_iter()
301	        .filter_map(|key| std::env::var_os(key).map(|value| (OsString::from(key), value)))
302	        .collect::<Vec<_>>();
303	    #[cfg(windows)]
304	    if let Ok(root) = windows_system_root() {
305	        environment.push((OsString::from("SystemRoot"), root));
306	    }
307	    environment
308	}

310	fn map_one_shot_completion(completion: OneShotCompletion) -> ToolchainCommandOutcome {
311	    match completion {
312	        OneShotCompletion::UnavailableBeforeDispatch => ToolchainCommandOutcome::Unavailable,
313	        OneShotCompletion::Succeeded => ToolchainCommandOutcome::Succeeded,
314	        OneShotCompletion::Failed => ToolchainCommandOutcome::Failed,
315	        OneShotCompletion::OutcomeUnknown => ToolchainCommandOutcome::Unknown,
316	    }
317	}

323	pub struct NativeToolchainRuntime {
324	    platform: ToolchainPlatform,

331	impl NativeToolchainRuntime {
332	    pub fn new(
333	        platform: ToolchainPlatform,
334	        arch: impl Into<String>,
335	        working_directory: PathBuf,
336	        uv_override: Option<PathBuf>,
337	        commands: Arc<dyn ToolchainCommandPort>,
338	    ) -> Self {
339	        Self {
340	            platform,
341	            arch: arch.into(),
342	            working_directory,
343	            uv_override,
344	            commands,
345	        }
346	    }
347	
348	    /// Builds the environment from the current process without exposing environment values.
349	    pub fn from_process(
350	        commands: Arc<dyn ToolchainCommandPort>,
351	    ) -> Result<Self, NativeToolchainRuntimeError> {
352	        let working_directory = std::env::current_dir()
353	            .map_err(|_| NativeToolchainRuntimeError::WorkingDirectoryUnavailable)?;
354	        let uv_override = std::env::var_os("MATCHACLAW_UV_BIN").map(PathBuf::from);
355	        Ok(Self::new(
356	            ToolchainPlatform::current(),
357	            std::env::consts::ARCH,
358	            working_directory,
359	            uv_override,
360	            commands,
361	        ))
362	    }
363	
364	    pub fn bundled_uv_available(&self) -> bool {
365	        self.bundled_uv_path_candidates()
366	            .into_iter()
367	            .any(|path| is_file(&path))
368	    }
369	
370	    pub fn bundled_uv_path_candidates(&self) -> Vec<PathBuf> {
371	        let executable = self.platform.bundled_uv_name();
372	        let target = format!("{}-{}", self.platform.target_name(), self.arch);
373	        let mut candidates = Vec::new();
374	
375	        if let Some(path) = self.uv_override.as_ref() {
376	            push_candidate(&mut candidates, path, &self.working_directory);
377	        }
378	        push_candidate(
379	            &mut candidates,
380	            &self
381	                .working_directory
382	                .join("resources")
383	                .join("bin")
384	                .join(&target)
385	                .join(executable),
386	            &self.working_directory,
387	        );
388	        push_candidate(
389	            &mut candidates,
390	            &self.working_directory.join("bin").join(executable),
391	            &self.working_directory,
392	        );
393	        candidates
394	    }
395	
396	    async fn uv_path(&self) -> Result<PathBuf, UvResolution> {
397	        if let Some(path) = self
398	            .bundled_uv_path_candidates()
399	            .into_iter()
400	            .find(|path| fs::metadata(path).is_ok_and(|metadata| metadata.is_file()))
401	        {
402	            return Ok(path);
403	        }
404	
405	        let request = ToolchainCommandRequest::new(
406	            self.platform.path_probe_name(),
407	            [OsString::from(UV_TOOL_ID)],
408	            Some(PATH_PROBE_TIMEOUT),
409	        );
410	        match self.commands.execute(request).await {
411	            ToolchainCommandOutcome::Succeeded => self
412	                .commands
413	                .resolve_program(Path::new(UV_TOOL_ID))
414	                .ok_or(UvResolution::Unavailable),
415	            ToolchainCommandOutcome::Failed | ToolchainCommandOutcome::Unavailable => {
416	                Err(UvResolution::Unavailable)
417	            }
418	            ToolchainCommandOutcome::TimedOut | ToolchainCommandOutcome::Unknown => {
419	                Err(UvResolution::Unknown)
420	            }
421	            ToolchainCommandOutcome::Unsupported => Err(UvResolution::Unsupported),
422	        }
423	    }
424	
425	    async fn observe(&self) -> ToolchainStatus {
426	        match self.uv_path().await {
427	            Ok(executable) => ToolchainStatus {
428	                uv: ToolAvailability::Available,
429	                python: self.observe_python(&executable).await,
430	            },
431	            Err(UvResolution::Unavailable) => ToolchainStatus {
432	                uv: ToolAvailability::Unavailable,
433	                python: PythonReadiness::Unavailable,
434	            },
435	            Err(UvResolution::Unknown) => ToolchainStatus {
436	                uv: ToolAvailability::Unknown,
437	                python: PythonReadiness::Unknown,
438	            },
439	            Err(UvResolution::Unsupported) => ToolchainStatus {
440	                uv: ToolAvailability::Unsupported,
441	                python: PythonReadiness::Unsupported,
442	            },
443	        }
444	    }
445	
446	    async fn observe_python(&self, executable: &Path) -> PythonReadiness {
447	        let request = ToolchainCommandRequest::new(
448	            executable.to_owned(),
449	            PYTHON_READINESS_ARGUMENTS.iter().map(OsString::from),
450	            Some(PATH_PROBE_TIMEOUT),
451	        );
452	        match self.commands.execute(request).await {
453	            ToolchainCommandOutcome::Succeeded => PythonReadiness::Ready,
454	            ToolchainCommandOutcome::Failed => PythonReadiness::NotReady,
455	            ToolchainCommandOutcome::Unavailable => PythonReadiness::Unavailable,
456	            ToolchainCommandOutcome::TimedOut | ToolchainCommandOutcome::Unknown => {
457	                PythonReadiness::Unknown
458	            }
459	            ToolchainCommandOutcome::Unsupported => PythonReadiness::Unsupported,
460	        }
461	    }
462	
463	    async fn install_uv(&self) -> UvInstallOutcome {
464	        let executable = match self.uv_path().await {
465	            Ok(path) => path,
466	            Err(UvResolution::Unavailable) => return UvInstallOutcome::Unavailable,
467	            Err(UvResolution::Unknown) => return UvInstallOutcome::Unknown,
468	            Err(UvResolution::Unsupported) => return UvInstallOutcome::Unsupported,
469	        };
470	        let install = ToolchainCommandRequest::new(
471	            executable.clone(),
472	            PYTHON_INSTALL_ARGUMENTS.iter().map(OsString::from),
473	            Some(INSTALL_TIMEOUT),
474	        );
475	        match self.commands.execute(install).await {
476	            ToolchainCommandOutcome::Succeeded => {
477	                let readiness = ToolchainCommandRequest::new(
478	                    executable,
479	                    PYTHON_READINESS_ARGUMENTS.iter().map(OsString::from),
480	                    Some(PATH_PROBE_TIMEOUT),
481	                );
482	                match self.commands.execute(readiness).await {
483	                    ToolchainCommandOutcome::Succeeded => UvInstallOutcome::Installed,
484	                    ToolchainCommandOutcome::Failed => UvInstallOutcome::Rejected,
485	                    ToolchainCommandOutcome::Unavailable => UvInstallOutcome::Unavailable,
486	                    ToolchainCommandOutcome::TimedOut | ToolchainCommandOutcome::Unknown => {
487	                        UvInstallOutcome::Unknown
488	                    }
489	                    ToolchainCommandOutcome::Unsupported => UvInstallOutcome::Unsupported,
490	                }
491	            }
492	            ToolchainCommandOutcome::Failed => UvInstallOutcome::Rejected,
493	            ToolchainCommandOutcome::Unavailable => UvInstallOutcome::Unavailable,
494	            ToolchainCommandOutcome::TimedOut | ToolchainCommandOutcome::Unknown => {
495	                UvInstallOutcome::Unknown
496	            }
497	            ToolchainCommandOutcome::Unsupported => UvInstallOutcome::Unsupported,
498	        }
499	    }
500	}

508	fn push_candidate(candidates: &mut Vec<PathBuf>, path: &Path, base: &Path) {
509	    let path = if path.is_absolute() {
510	        path.to_owned()
511	    } else {
512	        base.join(path)
513	    };
514	    if !candidates.iter().any(|candidate| candidate == &path) {
515	        candidates.push(path);
516	    }
517	}

669	pub struct ToolchainInstallOperation {
670	    job_id: String,
671	    snapshot: Arc<Mutex<ToolchainJobSnapshot>>,
672	    operation: Option<OperationHandle<UvInstallOutcome>>,
673	}
674	
675	impl ToolchainInstallOperation {
676	    fn submit(
677	        runtime: Arc<NativeToolchainRuntime>,
678	        sequence: u64,
679	        completion_notify: Arc<Notify>,
680	    ) -> Self {
681	        let job_id = format!("{TOOLCHAIN_JOB_ID_PREFIX}{sequence}");
682	        let snapshot = ToolchainJobSnapshot {
683	            id: job_id.clone(),
684	            job_type: TOOLCHAIN_JOB_TYPE.to_owned(),
685	            status: ToolchainJobStatus::Queued,
686	            queued_at: now_millis(),
687	            started_at: None,
688	            finished_at: None,
689	            attempts: 0,
690	            max_attempts: 1,
691	            progress: None,
692	            result: None,
693	            error: None,
694	        };
695	        let snapshot_state = Arc::new(Mutex::new(snapshot));
696	        let task_snapshot = Arc::clone(&snapshot_state);
697	        let (operation, _) = OperationHandle::spawn(move |cancellation| async move {
698	            {
699	                let mut snapshot = task_snapshot.lock().await;
700	                snapshot.status = ToolchainJobStatus::Running;
701	                snapshot.attempts = 1;
702	                snapshot.started_at = Some(now_millis());
703	                snapshot.progress = Some(ToolchainJobProgress {
704	                    updated_at: now_millis(),
705	                    percent: Some(10),
706	                    message: Some("Installing uv Python runtime".to_owned()),
707	                });
708	            }
709	            let outcome = tokio::select! {
710	                _ = cancellation.cancelled() => UvInstallOutcome::Unknown,
711	                outcome = runtime.install_uv() => outcome,
712	            };
713	            completion_notify.notify_waiters();
714	            outcome
715	        });
716	        Self { job_id, snapshot: snapshot_state, operation: Some(operation) }
717	    }
718	
719	    pub fn job_id(&self) -> &str { &self.job_id }
720	
721	    pub async fn snapshot(&self) -> ToolchainJobSnapshot {
722	        self.snapshot.lock().await.clone()
723	    }
724	
725	    pub async fn settle(&mut self) -> ToolchainJobSnapshot {
726	        let outcome = self.operation.take().map(|mut operation| async move {
727	            operation.join().await.ok().unwrap_or(UvInstallOutcome::Unknown)
728	        });
729	        if let Some(outcome) = outcome {
730	            let outcome = outcome.await;
731	            let mut snapshot = self.snapshot.lock().await;
732	            snapshot.status = match outcome {
733	                UvInstallOutcome::Installed | UvInstallOutcome::Rejected | UvInstallOutcome::Unknown => ToolchainJobStatus::Succeeded,
734	                UvInstallOutcome::Unavailable | UvInstallOutcome::Unsupported => ToolchainJobStatus::Failed,
735	            };
736	            snapshot.finished_at = Some(now_millis());
737	            snapshot.progress = Some(ToolchainJobProgress { updated_at: now_millis(), percent: Some(100), message: None });
738	            snapshot.result = Some(match outcome {
739	                UvInstallOutcome::Installed => ToolchainJobResult::Installed,
740	                UvInstallOutcome::Rejected => ToolchainJobResult::Rejected,
741	                UvInstallOutcome::Unknown => ToolchainJobResult::Unknown,
742	                UvInstallOutcome::Unavailable | UvInstallOutcome::Unsupported => ToolchainJobResult::Unavailable,
743	            });
744	            if matches!(outcome, UvInstallOutcome::Unavailable | UvInstallOutcome::Unsupported) {
745	                snapshot.error = Some(TOOLCHAIN_UNAVAILABLE_ERROR.to_owned());
746	            }
747	        }
748	        self.snapshot().await
749	    }
750	
751	    pub async fn cancel_and_join(&mut self) -> ToolchainJobSnapshot {
752	        if let Some(mut operation) = self.operation.take() {
753	            let _ = operation.cancel_and_join().await;
754	            let mut snapshot = self.snapshot.lock().await;
755	            snapshot.status = ToolchainJobStatus::Failed;
756	            snapshot.finished_at = Some(now_millis());
757	            snapshot.error = Some(TOOLCHAIN_CANCELLED_ERROR.to_owned());
758	        }
759	        self.snapshot().await
760	    }
761	}
762	
763	fn now_millis() -> u64 {
764	    SystemTime::now().duration_since(UNIX_EPOCH).ok().and_then(|duration| u64::try_from(duration.as_millis()).ok()).unwrap_or(u64::MAX)
765	}

797	pub struct OpenClawToolchain {
798	    state_dir: CanonicalStateDir,
799	    runtime: Arc<NativeToolchainRuntime>,
800	    install_operation: Arc<Mutex<Option<ToolchainInstallOperation>>>,
801	    completion_notify: Arc<Notify>,
802	    next_job_sequence: AtomicU64,
803	}
804	
805	impl OpenClawToolchain {
806	    pub fn new(state_dir: CanonicalStateDir, runtime: NativeToolchainRuntime) -> Self {
807	        Self {
808	            state_dir,
809	            runtime: Arc::new(runtime),
810	            install_operation: Arc::new(Mutex::new(None)),
811	            completion_notify: Arc::new(Notify::new()),
812	            next_job_sequence: AtomicU64::new(1),
813	        }
814	    }
815	
816	    pub async fn read(&self) -> ToolchainStatus {
817	        self.status().await
818	    }
819	
820	    pub async fn status(&self) -> ToolchainStatus {
821	        self.runtime.observe().await
822	    }
823	
824	    pub async fn list(&self) -> ToolchainCatalog {
825	        let status = self.status().await;
826	        ToolchainCatalog {
827	            entries: vec![
828	                ToolchainEntry {
829	                    id: UV_TOOL_ID.to_owned(),
830	                    status: status.uv,
831	                },
832	                ToolchainEntry {
833	                    id: PYTHON_TOOL_ID.to_owned(),
834	                    status: match status.python {
835	                        PythonReadiness::Ready => ToolAvailability::Available,
836	                        PythonReadiness::NotReady => ToolAvailability::Unavailable,
837	                        PythonReadiness::Unknown => ToolAvailability::Unknown,
838	                        PythonReadiness::Unavailable => ToolAvailability::Unavailable,
839	                        PythonReadiness::Unsupported => ToolAvailability::Unsupported,
840	                    },
841	                },
842	            ],
843	        }
844	    }
845	
846	    pub fn read_permission(&self) -> PermissionReadOutcome {
847	        match tool_permission::Mode::read(self.state_dir.clone()) {
848	            Ok(mode) => PermissionReadOutcome::Observed(mode),
849	            Err(tool_permission::Error::Unavailable) => PermissionReadOutcome::Unavailable,
850	            Err(tool_permission::Error::Unknown) => PermissionReadOutcome::Unknown,
851	        }
852	    }
853	
854	    pub fn write_permission(&self, mode: tool_permission::Mode) -> PermissionWriteOutcome {
855	        match mode.apply(self.state_dir.clone()) {
856	            Ok(tool_permission::Effect::Unchanged) => PermissionWriteOutcome::Unchanged,
857	            Ok(tool_permission::Effect::Written) => PermissionWriteOutcome::Written,
858	            Err(tool_permission::Error::Unavailable) => PermissionWriteOutcome::Unavailable,
859	            Err(tool_permission::Error::Unknown) => PermissionWriteOutcome::Unknown,
860	        }
861	    }
862	
863	    pub async fn install_uv(&self) -> UvInstallOutcome {
864	        self.runtime.install_uv().await
865	    }
866	
867	    pub async fn submit_install(&self) -> ToolchainJobSubmission {
868	        let mut operation = self.install_operation.lock().await;
869	        if let Some(existing) = operation.as_ref() {
870	            let snapshot = existing.snapshot().await;
871	            if matches!(snapshot.status, ToolchainJobStatus::Queued | ToolchainJobStatus::Running) {
872	                return ToolchainJobSubmission { job: snapshot };
873	            }
874	        }
875	        let sequence = self.next_job_sequence.fetch_add(1, Ordering::Relaxed);
876	        let new_operation = ToolchainInstallOperation::submit(
877	            Arc::clone(&self.runtime),
878	            sequence,
879	            Arc::clone(&self.completion_notify),
880	        );
881	        let snapshot = new_operation.snapshot().await;
882	        let notify = Arc::clone(&self.completion_notify);
883	        let operation_state = Arc::clone(&self.install_operation);
884	        *operation = Some(new_operation);
885	        tokio::spawn(async move {
886	            loop {
887	                let finished = {
888	                    let operation = operation_state.lock().await;
889	                    operation.as_ref().is_some_and(|operation| {
890	                        operation.operation.as_ref().is_some_and(|task| task.is_finished())
891	                    })
892	                };
893	                if finished {
894	                    notify.notify_waiters();
895	                    break;
896	                }
897	                tokio::time::sleep(Duration::from_millis(25)).await;
898	            }
899	        });
900	        ToolchainJobSubmission { job: snapshot }
901	    }
902	
903	    pub async fn job_get(&self, job_id: &str) -> ToolchainJobLookup {
904	        let operation = self.install_operation.lock().await;
905	        match operation.as_ref() {
906	            Some(operation) if operation.job_id() == job_id => {
907	                ToolchainJobLookup::Known(operation.snapshot().await)
908	            }
909	            _ => ToolchainJobLookup::Unknown,
910	        }
911	    }
912	
913	    pub async fn wait_for_install_completion(&self) {
914	        loop {
915	            let notified = self.completion_notify.notified();
916	            let finished = {
917	                let operation = self.install_operation.lock().await;
918	                operation.as_ref().is_some_and(|operation| {
919	                    operation.operation.as_ref().is_some_and(|task| task.is_finished())
920	                })
921	            };
922	            if finished {
923	                return;
924	            }
925	            notified.await;
926	        }
927	    }
928	
929	    pub async fn reap_install(&self) -> Option<ToolchainJobSnapshot> {
930	        let mut operation = self.install_operation.lock().await;
931	        let operation_ref = operation.as_mut()?;
932	        if !operation_ref.operation.as_ref().is_some_and(|operation| operation.is_finished()) {
933	            return None;
934	        }
935	        Some(operation_ref.settle().await)
936	    }
937	
938	    pub async fn cancel_install(&self) -> Option<ToolchainJobSnapshot> {
939	        let mut operation = self.install_operation.lock().await;
940	        Some(operation.as_mut()?.cancel_and_join().await)
941	    }
942	
943	    pub async fn write(&self, request: ToolchainWriteRequest) -> ToolchainWriteOutcome {
944	        match request {
945	            ToolchainWriteRequest::Permission(mode) => {
946	                ToolchainWriteOutcome::Permission(self.write_permission(mode))
947	            }
948	            ToolchainWriteRequest::InstallUv => {
949	                ToolchainWriteOutcome::InstallUv(self.install_uv().await)
950	            }
951	        }
952	    }
953	}
